cmake_minimum_required(VERSION 3.8)
project(thunder_controllers LANGUAGES CXX)

if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

# find dependencies
find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(controller_interface REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(generate_parameter_library REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(realtime_tools REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(yaml-cpp REQUIRED)

set(THIS_PACKAGE_INCLUDE_DEPENDS
  ament_index_cpp
  controller_interface
  generate_parameter_library
  hardware_interface
  pluginlib
  rclcpp
  rclcpp_lifecycle
  realtime_tools
  sensor_msgs
)

# ------------------------------------------------------------
#   Thunder-generated robot model
# ------------------------------------------------------------
# Copy the folder generated by `thunder gen` into 'thunder/':
#   thunder/<robot>_generatedFiles/{thunder_<robot>.h, thunder_<robot>.cpp, <robot>_gen.h, <robot>_gen.cpp, <robot>_par.yaml}
set(THUNDER_ROBOT "Rizon4s" CACHE STRING "Name of the Thunder-generated robot (class thunder_<robot>)")
set(THUNDER_DIR ${CMAKE_CURRENT_SOURCE_DIR}/thunder/${THUNDER_ROBOT}_generatedFiles)
set(THUNDER_SOURCES
  ${THUNDER_DIR}/thunder_${THUNDER_ROBOT}.cpp
  ${THUNDER_DIR}/${THUNDER_ROBOT}_gen.cpp
)

foreach(file ${THUNDER_SOURCES} ${THUNDER_DIR}/thunder_${THUNDER_ROBOT}.h)
  if(NOT EXISTS ${file})
    message(WARNING
      "thunder_controllers: missing Thunder file '${file}'. "
      "Generate the '${THUNDER_ROBOT}' model with Thunder and copy it into ${THUNDER_DIR}. "
      "Skipping the controllers build.")
    ament_package()
    return()
  endif()
endforeach()

add_library(thunder_${THUNDER_ROBOT} STATIC ${THUNDER_SOURCES})
set_target_properties(thunder_${THUNDER_ROBOT} PROPERTIES POSITION_INDEPENDENT_CODE ON)
# The CasADi-generated source is huge: -O1 keeps compile time and memory reasonable
set_source_files_properties(${THUNDER_DIR}/${THUNDER_ROBOT}_gen.cpp PROPERTIES COMPILE_OPTIONS "-O1")
target_include_directories(thunder_${THUNDER_ROBOT} SYSTEM PUBLIC ${THUNDER_DIR} ${EIGEN3_INCLUDE_DIR})
target_link_libraries(thunder_${THUNDER_ROBOT} PUBLIC yaml-cpp)

# ------------------------------------------------------------
#   Controllers
# ------------------------------------------------------------
generate_parameter_library(thunder_computed_torque_controller_parameters
  config/thunder_computed_torque_controller_parameters.yaml
)

add_library(${PROJECT_NAME}
  SHARED
  src/thunder_computed_torque_controller.cpp
)

target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(${PROJECT_NAME} PRIVATE
  THUNDER_ROBOT_HEADER="thunder_${THUNDER_ROBOT}.h"
  THUNDER_ROBOT_CLASS=thunder_${THUNDER_ROBOT}
  THUNDER_ROBOT_NAME="${THUNDER_ROBOT}"
)

target_include_directories(${PROJECT_NAME} PRIVATE include)

target_link_libraries(${PROJECT_NAME} PUBLIC
  thunder_computed_torque_controller_parameters
  thunder_${THUNDER_ROBOT}
)

ament_target_dependencies(${PROJECT_NAME} PUBLIC
  ${THIS_PACKAGE_INCLUDE_DEPENDS}
)

# prevent pluginlib from using boost
target_compile_definitions(${PROJECT_NAME} PUBLIC "PLUGINLIB__DISABLE_BOOST_FUNCTIONS")
pluginlib_export_plugin_description_file(controller_interface thunder_controllers.xml)

install(
  DIRECTORY include/
  DESTINATION include/${PROJECT_NAME}
)

install(
  TARGETS
    ${PROJECT_NAME}
    thunder_computed_torque_controller_parameters
  EXPORT export_${PROJECT_NAME}
  ARCHIVE DESTINATION lib
  LIBRARY DESTINATION lib
  RUNTIME DESTINATION bin
)

# ------------------------------------------------------------
#   Model check node (Thunder model vs Gazebo joint torque sensors)
# ------------------------------------------------------------
add_executable(thunder_model_check src/thunder_model_check.cpp)
target_compile_options(thunder_model_check PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(thunder_model_check PRIVATE
  THUNDER_ROBOT_HEADER="thunder_${THUNDER_ROBOT}.h"
  THUNDER_ROBOT_CLASS=thunder_${THUNDER_ROBOT}
)
target_link_libraries(thunder_model_check thunder_${THUNDER_ROBOT})
ament_target_dependencies(thunder_model_check ament_index_cpp geometry_msgs rclcpp sensor_msgs)
install(TARGETS thunder_model_check DESTINATION lib/${PROJECT_NAME})

# ------------------------------------------------------------
#   Test: evaluate the Thunder model at a given (q, dq, ddq)
# ------------------------------------------------------------
add_executable(thunder_dynamics_test test/thunder_dynamics_test.cpp)
target_compile_options(thunder_dynamics_test PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(thunder_dynamics_test PRIVATE
  THUNDER_ROBOT_HEADER="thunder_${THUNDER_ROBOT}.h"
  THUNDER_ROBOT_CLASS=thunder_${THUNDER_ROBOT}
  THUNDER_ROBOT_NAME="${THUNDER_ROBOT}"
)
target_link_libraries(thunder_dynamics_test thunder_${THUNDER_ROBOT})
ament_target_dependencies(thunder_dynamics_test ament_index_cpp)
install(TARGETS thunder_dynamics_test DESTINATION lib/${PROJECT_NAME})

# Thunder parameter files (e.g. <robot>_par.yaml) and example controller config
install(
  FILES ${THUNDER_DIR}/${THUNDER_ROBOT}_par.yaml
  DESTINATION share/${PROJECT_NAME}/thunder
  OPTIONAL
)
install(
  DIRECTORY config
  DESTINATION share/${PROJECT_NAME}
)

ament_export_dependencies(${THIS_PACKAGE_INCLUDE_DEPENDS})
ament_export_include_directories(include)
ament_export_libraries(${PROJECT_NAME})

ament_package()
